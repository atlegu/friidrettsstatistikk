% Tittelblokk: hovedtittel / undertittel / byline med linjaler
\newcommand{\artikkeltittel}[4]{%
  \thispagestyle{empty}%
  {\raggedright\sffamily\bfseries\fontsize{30}{34}\selectfont\color{morkgra}#1\par}
  \vspace{8pt}
  {\raggedright\sffamily\fontsize{17}{22}\selectfont\color{hovedbla}#2\par}
  \vspace{16pt}
  {\noindent\color{hovedbla}\rule{\textwidth}{2.2pt}\par}
  \vspace{7pt}
  {\raggedright\sffamily\footnotesize\color{lysgra}%
   \MakeUppercase{#3}\hfill\MakeUppercase{#4}\par}
  \vspace{3pt}
  {\noindent\color{lysgra!50}\rule{\textwidth}{0.5pt}\par}
  \vspace{18pt}
}

% Ingress: større kursiv, mørk grå, luftig

   \setlength{\rightskip}{1.2cm plus 1fil}}
  {\par\endgroup\vspace{12pt}}
